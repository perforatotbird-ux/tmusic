│   │   └── utils/               # like, http, client_ip, msg_ids, time
│   ├── tests/                   # pytest
│   └── requirements.txt
├── frontend/
│   ├── src/
│   │   ├── api/                 # client, auth, library, stream
│   │   ├── components/          # MusicLibrary, Player, AuthModal, CoverPicker, AlbumDetail
│   │   ├── store/               # authStore, playerStore
│   │   ├── services/            # mediaSession, coverMissing
│   │   ├── utils/               # cover
│   │   ├── App.jsx
│   │   └── main.jsx
│   ├── public/                  # icon.svg, manifest.json (PWA)
│   ├── dist/                    # production bundle
│   ├── package.json
│   └── vite.config.js

│   └── android/                 # Нативный Android проект (FGS, MediaSession)
├── scripts/
│   ├── build_android_apk.ps1    # Сборка debug/release APK для Android
│   ├── migrate_data.py
│   ├── create_session.py
│   ├── secure_tg_accounts.py
│   └── init_admin.py
├── cache/                       # music_covers, music_images

├── Dockerfile / docker-compose.yml
